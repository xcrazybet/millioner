// ===== ADD THIS AT THE TOP OF YOUR SCRIPT SECTION ===== // ===== BET SETTLEMENT SERVICE ===== // This runs every 60 seconds to check and settle bets function startBetSettlementService() { setInterval(async () => { if (!auth.currentUser) return; try { // Get all pending bets const snapshot = await db.collection("football_bets") .where("status", "==", "pending") .get(); for (const doc of snapshot.docs) { const bet = doc.data(); // Check if all matches are finished const allFinished = await checkMatchesFinished(bet.selections); if (allFinished) { await settleBet(bet, doc.id); } } } catch (error) { console.error('Error in bet settlement:', error); } }, 60000); // Check every minute } // Check if all matches in a bet are finished async function checkMatchesFinished(selections) { try { // Get unique match IDs const matchIds = [...new Set(selections.map(s => s.matchId))]; // Fetch match statuses from API for (const matchId of matchIds) { const response = await fetch(`${BASE_URL}/fixtures?id=${matchId}`, { headers: { 'x-apisports-key': API_KEY } }); if (response.ok) { const data = await response.json(); const match = data.response[0]; if (!match || !['FT', 'AET', 'PEN'].includes(match.fixture.status.short)) { return false; } } } return true; } catch (error) { console.error('Error checking matches:', error); return false; } } // Settle a single bet async function settleBet(bet, betId) { try { // Check if bet wins const won = await checkBetWins(bet.selections); let winAmount = 0; let status = 'lost'; if (won) { winAmount = bet.stake * bet.totalOdds; status = 'won'; // Credit winnings to user await processTransaction( winAmount, "football_win", `Win #${bet.ticketNumber}`, bet.userId ); } // Update bet status await db.collection("football_bets").doc(betId).update({ status: status, return: won ? winAmount : 0, settledAt: new Date(), result: won ? 'win' : 'loss' }); // Notify if current user if (bet.userId === auth.currentUser?.uid) { showToast( won ? `🎉 Bet won! +$${winAmount.toFixed(2)}` : '😞 Bet lost', won ? 'success' : 'error' ); } console.log(`Bet ${bet.ticketNumber} settled as ${status}`); } catch (error) { console.error('Error settling bet:', error); } } // Check if selections win async function checkBetWins(selections) { for (const selection of selections) { try { // Fetch match result const response = await fetch(`${BASE_URL}/fixtures?id=${selection.matchId}`, { headers: { 'x-apisports-key': API_KEY } }); if (!response.ok) return false; const data = await response.json(); const match = data.response[0]; if (!match) return false; const homeGoals = match.goals.home; const awayGoals = match.goals.away; const totalGoals = homeGoals + awayGoals; // Check if selection wins let selectionWon = false; switch(selection.market) { case 'Match Winner': if (selection.selection === '1') selectionWon = homeGoals > awayGoals; else if (selection.selection === 'X') selectionWon = homeGoals === awayGoals; else if (selection.selection === '2') selectionWon = awayGoals > homeGoals; break; case 'Total Goals': const threshold = parseFloat(selection.display.split(' ')[1]); if (selection.selection.includes('Over')) selectionWon = totalGoals > threshold; else if (selection.selection.includes('Under')) selectionWon = totalGoals < threshold; break; case 'BTTS': const bothScored = homeGoals > 0 && awayGoals > 0; selectionWon = bothScored === (selection.selection === 'Yes'); break; } if (!selectionWon) return false; } catch (error) { console.error('Error checking selection:', error); return false; } } return true; } // Enhanced transaction processor with user ID async function processTransaction(amount, type, description, userId) { try { const walletRef = db.collection("wallets").doc(userId); return await db.runTransaction(async (transaction) => { const walletDoc = await transaction.get(walletRef); if (!walletDoc.exists) { throw new Error("Wallet not found"); } const currentBalance = walletDoc.data().balance || 0; const newBalance = currentBalance + amount; if (newBalance < 0) { throw new Error("Insufficient funds"); } transaction.update(walletRef, { balance: newBalance, updatedAt: new Date() }); const transactionRef = walletRef.collection("transactions").doc(); transaction.set(transactionRef, { type: type, amount: amount, description: description, createdAt: new Date(), beforeBalance: currentBalance, afterBalance: newBalance }); return { success: true, newBalance }; }); } catch (error) { console.error('Transaction failed:', error); return { success: false, error: error.message }; } } // Update the placeBet function to ensure proper Firebase storage async function placeBet() { const betAmount = parseFloat(document.getElementById('bet-amount').value); // Validation if (gameState.selectedBets.length === 0) { showToast('Select at least one bet', 'error'); return; } if (betAmount < BETTING_RULES.minBet || betAmount > BETTING_RULES.maxBet) { showToast(`Bet must be between $${BETTING_RULES.minBet} and $${BETTING_RULES.maxBet}`, 'error'); return; } if (betAmount > gameState.balance) { showToast('Insufficient funds', 'error'); return; } if (gameState.userLimits.todayWagered + betAmount > gameState.userLimits.dailyMax) { showToast('Daily limit exceeded', 'error'); return; } if (gameState.isPlacingBet) return; gameState.isPlacingBet = true; document.getElementById('place-bet-btn').disabled = true; const totalOdds = gameState.selectedBets.reduce((acc, bet) => acc * bet.odds, 1); const potentialWin = betAmount * totalOdds; const ticketNumber = generateTicketNumber(); try { // Process bet transaction const result = await processTransaction( -betAmount, "football_bet", `Bet #${ticketNumber} - ${gameState.selectedBets.length} selections @ ${totalOdds.toFixed(2)}`, gameState.currentUser.uid ); if (!result.success) { throw new Error(result.error); } // Create bet object with proper structure const bet = { ticketNumber, userId: gameState.currentUser.uid, selections: gameState.selectedBets.map(s => ({ matchId: s.matchId, matchName: s.matchName, market: s.market, selection: s.selection, display: s.display, odds: s.odds })), stake: betAmount, totalOdds, potentialWin, betType: gameState.betType, systemType: gameState.betType === 'system' ? gameState.systemType : null, eachWay: gameState.eachWay, status: 'pending', createdAt: firebase.firestore.FieldValue.serverTimestamp() }; // Save to Firebase with ticket number as document ID await db.collection("football_bets").doc(ticketNumber).set(bet); // Update local state gameState.balance = result.newBalance; updateBalanceDisplay(); gameState.userLimits.todayWagered += betAmount; gameState.stats.totalBets++; gameState.stats.totalStaked += betAmount; showToast(`Bet placed! Ticket #${ticketNumber}`, 'success'); // Clear bet slip gameState.selectedBets = []; updateBetSlip(); // Refresh open bets await loadOpenBets(); } catch (error) { console.error('Error placing bet:', error); showToast('Failed to place bet: ' + error.message, 'error'); } finally { gameState.isPlacingBet = false; document.getElementById('place-bet-btn').disabled = false; } } // Update loadOpenBets to use proper ordering async function loadOpenBets() { try { const snapshot = await db.collection("football_bets") .where("userId", "==", gameState.currentUser.uid) .where("status", "==", "pending") .orderBy("createdAt", "desc") .get(); gameState.openBets = []; snapshot.forEach(doc => { const data = doc.data(); gameState.openBets.push({ ...data, id: doc.id, createdAt: data.createdAt?.toDate() || new Date() }); }); displayOpenBets(); } catch (error) { console.error('Error loading open bets:', error); } } // Update loadBetHistory with proper ordering async function loadBetHistory() { try { const snapshot = await db.collection("football_bets") .where("userId", "==", gameState.currentUser.uid) .orderBy("createdAt", "desc") .limit(50) .get(); gameState.betHistory = []; snapshot.forEach(doc => { const data = doc.data(); gameState.betHistory.push({ ...data, id: doc.id, timestamp: data.createdAt?.toDate() || new Date() }); }); displayHistory(); updateStats(); } catch (error) { console.error('Error loading history:', error); } } // Start the bet settlement service after auth auth.onAuthStateChanged((user) => { if (user) { gameState.currentUser = user; loadBalance(); loadMatches(); loadOpenBets(); loadBetHistory(); loadUserLimits(); setupRealTimeListeners(); startBetSettlementService(); // Start the settlement service } else { showToast('Please login to continue', 'error'); setTimeout(() => { window.location.href = 'login.html'; }, 2000); } });
